# GitHub's remote MCP server — repositories, issues, pull requests and actions
# as agent tools:
#   { "mcpServers": { "github": { "url": "https://api.githubcopilot.com/mcp/" } } }
#
# This exists as its own profile because of a naming trap. The server is hosted
# on a *Copilot* domain, so without this you would have to enable the whole
# `copilot` profile to use GitHub's MCP server — opting into sending your code
# to GitHub for completions, which is a separate decision you may not want to
# make. This is the narrower grant: the MCP endpoint and nothing else.
#
# Use it alongside `base`, which covers github.com itself.
api.githubcopilot.com
